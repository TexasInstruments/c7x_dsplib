/*
 *  Sends the test parameter structure and number of tests
 */
void DSPLIB_svd_small_getTestParams (
    DSPLIB_svd_small_testParams_t **params,
    int32_t *                                 numTests)
{
   *params = testParams;
   *numTests =
       sizeof (testParams) / sizeof (DSPLIB_svd_small_testParams_t);
}
